<script id="vl-alys-safe-rail">
(function(){
  const GAP=8;
  const protectedSelector='a,button,input,textarea,select,p,h1,h2,h3,h4,li,td,th,span,em,b,i,small,figcaption,[role="button"],[role="link"]';
  let raf=0, observer=null;
  function visible(el){
    if(!el || el.classList.contains('vl-aifab') || el.closest('.vl-aipanel')) return false;
    const cs=getComputedStyle(el);
    if(cs.display==='none'||cs.visibility==='hidden'||Number(cs.opacity)===0) return false;
    const r=el.getBoundingClientRect();
    return r.width>1&&r.height>1&&r.bottom>0&&r.right>0&&r.top<innerHeight&&r.left<innerWidth;
  }
  function intersects(a,b){return !(a.right<=b.left-GAP||a.left>=b.right+GAP||a.bottom<=b.top-GAP||a.top>=b.bottom+GAP);}
  function rectAt(x,y,w,h){return {left:x,top:y,right:x+w,bottom:y+h,width:w,height:h};}
  function currentScreen(){
    const screens=[...document.querySelectorAll('.vl-screen')].filter(visible);
    if(!screens.length) return document.body;
    let best=screens[0],score=-1;
    for(const s of screens){const r=s.getBoundingClientRect();const inter=Math.max(0,Math.min(r.bottom,innerHeight)-Math.max(r.top,0));if(inter>score){score=inter;best=s;}}
    return best;
  }
  function place(){
    raf=0;
    const fab=document.querySelector('.vl-aifab'); if(!fab) return;
    fab.classList.add('vl-aifab-safe');
    fab.style.left='auto';fab.style.top='auto';fab.style.right='18px';fab.style.setProperty('bottom','calc(18px + env(safe-area-inset-bottom,0px))','important');
    const fr=fab.getBoundingClientRect(); const w=Math.ceil(fr.width||52),h=Math.ceil(fr.height||52);
    const screen=currentScreen();
    const protectedEls=[...document.querySelectorAll(protectedSelector)]
      .filter(visible)
      .filter(el=>!el.contains(fab))
      // Decorative wrappers such as empty scrims must not block the entire viewport.
      .filter(el=>!['SPAN','EM','B','I','SMALL','FIGCAPTION'].includes(el.tagName) || (el.textContent||'').trim().length>0);
    const boxes=protectedEls.map(el=>el.getBoundingClientRect()).filter(r=>r.bottom>0&&r.top<innerHeight);
    const header=document.querySelector('header'); const hr=header&&visible(header)?header.getBoundingClientRect():null;
    const minY=Math.max(8,hr?Math.min(innerHeight- h-8,hr.bottom+8):8);
    const maxY=Math.max(minY,innerHeight-h-8);
    const xs=[];
    const rightX=Math.max(8,innerWidth-w-8), leftX=8;
    xs.push(rightX); if(leftX!==rightX) xs.push(leftX);
    // If both edge rails are occupied, search interior visual whitespace before collapsing.
    // Right-biased order preserves the familiar Alys position whenever possible.
    if(innerWidth<=700){
      const usable=Math.max(0,innerWidth-w-16);
      for(const ratio of [0.82,0.66,0.50,0.34,0.18]){
        const x=Math.max(8,Math.min(rightX,8+Math.round(usable*ratio)));
        if(!xs.includes(x)) xs.push(x);
      }
    } else {
      xs.push(Math.max(8,innerWidth-w-24));
    }
    let candidates=[];
    for(const x of xs){for(let y=maxY;y>=minY;y-=10)candidates.push([x,y]);}
    // Prefer the normal bottom-right location, then nearest safe edge slot.
    candidates.sort((a,b)=>{const da=(rightX-a[0])**2+(maxY-a[1])**2;const db=(rightX-b[0])**2+(maxY-b[1])**2;return da-db;});
    let chosen=null;
    outer: for(const [x,y] of candidates){
      const rr=rectAt(x,y,w,h);
      for(const b of boxes){if(intersects(rr,b)) continue outer;}
      chosen=[x,y];break;
    }
    if(chosen){fab.style.right='auto';fab.style.setProperty('bottom','auto','important');fab.style.left=chosen[0]+'px';fab.style.top=chosen[1]+'px';fab.style.opacity='1';fab.style.visibility='visible';fab.style.pointerEvents='auto';}
    else {
      // Dense viewport fallback: collapse the floating entry rather than cover meaningful content.
      // It reappears automatically at the next safe scroll position; Alys logic/panel are unchanged.
      fab.style.opacity='0';fab.style.visibility='hidden';fab.style.pointerEvents='none';
    }
  }
  function schedule(){if(!raf)raf=requestAnimationFrame(place);}
  function boot(){
    schedule(); addEventListener('scroll',schedule,{passive:true}); addEventListener('resize',schedule,{passive:true});
    observer=new MutationObserver(schedule); observer.observe(document.body,{subtree:true,childList:true}); setTimeout(schedule,120); setTimeout(schedule,650); if(document.fonts&&document.fonts.ready) document.fonts.ready.then(schedule).catch(()=>{});
  }
  if(document.readyState==='loading')document.addEventListener('DOMContentLoaded',boot,{once:true});else boot();
})();
